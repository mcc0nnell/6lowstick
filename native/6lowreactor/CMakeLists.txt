cmake_minimum_required(VERSION 3.18)
project(sixlow_reactor VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
option(S6_WITH_CELIX "Build optional Celix mock service bundles and demo container" OFF)

find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(s6_reactor STATIC src/s6_reactor.c)
set_target_properties(s6_reactor PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(s6_reactor PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(s6_reactor PUBLIC OpenSSL::Crypto)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(s6_reactor PRIVATE -Wall -Wextra -Werror)
endif()

enable_testing()
add_executable(s6_reactor_test test/reactor_test.c)
target_link_libraries(s6_reactor_test PRIVATE s6_reactor)
add_test(NAME sixlow_reactor_contract COMMAND s6_reactor_test)

if(S6_WITH_CELIX)
  find_package(Celix REQUIRED)
  add_celix_bundle(s6_radio_thread SOURCES src/mock_radio.c)
  target_compile_definitions(s6_radio_thread PRIVATE S6_MOCK_ROUTE="thread" S6_MOCK_ACK=0)
  target_link_libraries(s6_radio_thread PRIVATE s6_reactor)
  add_celix_bundle(s6_radio_lora SOURCES src/mock_radio.c)
  target_compile_definitions(s6_radio_lora PRIVATE S6_MOCK_ROUTE="lora-p2p" S6_MOCK_ACK=1)
  target_link_libraries(s6_radio_lora PRIVATE s6_reactor)
  add_celix_bundle(s6_router SOURCES src/router_bundle.c)
  target_link_libraries(s6_router PRIVATE s6_reactor)
  add_celix_container(s6_reactor_demo C BUNDLES s6_radio_thread s6_radio_lora s6_router)
  # Negative control: absent LoRa service may never be interpreted as a receipt.
  add_celix_container(s6_reactor_no_lora C BUNDLES s6_radio_thread s6_router)
endif()
