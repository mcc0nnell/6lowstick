cmake_minimum_required(VERSION 3.16)
project(6lowbang LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
add_library(lowbang STATIC src/lowbang.c)
target_include_directories(lowbang PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lowbang PUBLIC OpenSSL::Crypto)
# 6s owns the binary signed-event envelope and bounded host-local
# development transport. Tronix/Celix may link these, not copy the codec.
add_library(lowbang_wire STATIC src/lowbang_wire.c)
target_include_directories(lowbang_wire PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lowbang_wire PUBLIC lowbang)
add_library(lowbang_udp STATIC src/lowbang_udp.c)
target_include_directories(lowbang_udp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lowbang_udp PUBLIC lowbang_wire)
set_target_properties(lowbang lowbang_wire lowbang_udp
  PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lowbang PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(lowbang_wire PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(lowbang_udp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(lowbang-test tests/test_lowbang.c)
target_link_libraries(lowbang-test PRIVATE lowbang)
add_executable(lowbang-demo examples/demo.c)
target_link_libraries(lowbang-demo PRIVATE lowbang)
enable_testing()
add_test(NAME bang-native-unit COMMAND lowbang-test)

add_executable(lowbang-transport-test tests/test_transport.c)
target_link_libraries(lowbang-transport-test PRIVATE lowbang_udp)
add_test(NAME bang-native-binary-transport COMMAND lowbang-transport-test)

add_executable(lowbang-posix-codec examples/posix_codec.c)
target_compile_definitions(lowbang-posix-codec PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(lowbang-posix-codec PRIVATE lowbang_wire)
add_test(NAME bang-native-posix-codec COMMAND lowbang-posix-codec selftest)
